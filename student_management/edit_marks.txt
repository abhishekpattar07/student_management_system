<?php
session_start();
require_once "config/db.php";
if (!isset($_SESSION["user_id"])) {
 header("Location: login.php");
 exit();
}
$id = intval($_GET["id"]);
$stmt = $conn->prepare(
 "SELECT * FROM marks WHERE id = ?"
);
$stmt->bind_param("i", $id);
$stmt->execute();
$result = $stmt->get_result();
$mark = $result->fetch_assoc();
if (!$mark) {
 die("Mark record not found.");
}
$student_id = $mark["student_id"];
if ($_SERVER["REQUEST_METHOD"] == "POST") {
 $subject = trim($_POST["subject"]);
 $marks = intval($_POST["marks"]);
 if ($marks < 0 || $marks > 100) {
 die("Marks must be between 0 and 100.");
 }
 $stmt = $conn->prepare(
 "UPDATE marks
 SET subject=?, marks=?
 WHERE id=?"
 );
 $stmt->bind_param(
 "sii",
 $subject, $marks, $id
 );
 $stmt->execute();
 header(
 "Location: marks.php?student_id=" . $student_id
 );
 exit();
}
?>
<!DOCTYPE html>
<html>
<head>
<title>Edit Marks</title>
<link rel="stylesheet" href="css/style.css">
</head>
<body>
<div class="container">
<div class="card">
<h2>Edit Marks</h2>
<form method="POST">
<label>Subject</label>
<input type="text" name="subject"
value="<?php echo htmlspecialchars($mark["subject"]); ?>"
required>
<label>Marks</label>
<input type="number" name="marks"
min="0" max="100"
value="<?php echo $mark["marks"]; ?>"
required>
<button type="submit">Update Marks</button>
</form>
<p>
<a href="marks.php?student_id=<?php echo $student_id; ?>">
← Back
</a>
</p>
</div>
</div>
</body>
</html>
